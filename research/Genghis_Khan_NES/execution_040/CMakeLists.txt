cmake_minimum_required(VERSION 3.15)
project(GK1Execution040 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(gk1_battle_snapshot STATIC gk1_battle_snapshot.c ../execution_035/gk1_native_ram_view.c)
target_include_directories(gk1_battle_snapshot PUBLIC . ../execution_035)
add_executable(test_gk1_snapshot test_gk1_snapshot.c)
target_link_libraries(test_gk1_snapshot PRIVATE gk1_battle_snapshot)
add_executable(gk1_snapshot_cli gk1_snapshot_cli.c)
target_link_libraries(gk1_snapshot_cli PRIVATE gk1_battle_snapshot)
enable_testing()
add_test(NAME GK1_execution_040 COMMAND test_gk1_snapshot)
