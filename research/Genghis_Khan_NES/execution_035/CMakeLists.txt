cmake_minimum_required(VERSION 3.16)
project(GK1NativeSRAMView C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(gk1_native_ram_view STATIC gk1_native_ram_view.c)
target_include_directories(gk1_native_ram_view PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(test_gk1_native_ram_view test_gk1_native_ram_view.c)
target_link_libraries(test_gk1_native_ram_view PRIVATE gk1_native_ram_view)
enable_testing()
add_test(NAME native_ram_view COMMAND test_gk1_native_ram_view)
