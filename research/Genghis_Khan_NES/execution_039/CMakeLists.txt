cmake_minimum_required(VERSION 3.15)
project(GK1Execution039 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(gk1_session STATIC gk1_battle_session.c ../execution_034/gk1_tactical_slice.c)
target_include_directories(gk1_session PUBLIC . ../execution_034)
add_executable(test_gk1_session test_gk1_session.c)

target_link_libraries(gk1_battle_replay PRIVATE gk1_session)
enable_testing()
add_test(NAME GK1_execution_039 COMMAND test_gk1_session)
