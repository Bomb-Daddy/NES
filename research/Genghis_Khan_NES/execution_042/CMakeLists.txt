cmake_minimum_required(VERSION 3.15)
project(GK1NativeShell042 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(gk1_app_state STATIC gk1_app_state.c ../execution_039/gk1_battle_session.c ../execution_034/gk1_tactical_slice.c)
target_include_directories(gk1_app_state PUBLIC . ../execution_039 ../execution_034)
add_library(gk1_replay_store STATIC gk1_replay_store.c)
target_link_libraries(gk1_replay_store PUBLIC gk1_app_state)
add_executable(test_gk1_app_state test_gk1_app_state.c)
target_link_libraries(test_gk1_app_state PRIVATE gk1_app_state)
if(WIN32)
  add_executable(gk1_native_shell WIN32 gk1_native_shell.c)
  target_link_libraries(gk1_native_shell PRIVATE gk1_replay_store user32 gdi32)
else()
  find_package(X11 REQUIRED)
  add_executable(gk1_native_shell gk1_native_shell.c)
  target_link_libraries(gk1_native_shell PRIVATE gk1_replay_store X11::X11)
endif()
add_executable(test_gk1_replay_store test_gk1_replay_store.c)
target_link_libraries(test_gk1_replay_store PRIVATE gk1_replay_store)
enable_testing()
add_test(NAME GK1_042_C11_ControllerState COMMAND test_gk1_app_state)

add_test(NAME GK1_042_Exclusive_Research_Replay COMMAND test_gk1_replay_store)
