cmake_minimum_required(VERSION 3.15)
project(GK61Market C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(gk61_market STATIC gk61_market.c)
target_include_directories(gk61_market PUBLIC .)
add_executable(test_gk61_market test_gk61_market.c)
target_link_libraries(test_gk61_market PRIVATE gk61_market)
enable_testing()
add_test(NAME GK61_OriginalMerchantQuotesAndTransactions COMMAND test_gk61_market)
