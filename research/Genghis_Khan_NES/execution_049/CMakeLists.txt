cmake_minimum_required(VERSION 3.15)
project(GK1ScenarioFactory049 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
add_library(gk1_scenario_factory STATIC gk1_scenario_factory.c)
target_include_directories(gk1_scenario_factory PUBLIC .)
add_executable(test_gk1_scenario_factory test_gk1_scenario_factory.c)
target_link_libraries(test_gk1_scenario_factory PRIVATE gk1_scenario_factory)
add_executable(gk1_scenario_list gk1_scenario_list.c)
target_link_libraries(gk1_scenario_list PRIVATE gk1_scenario_factory)
enable_testing()
add_test(NAME GK1_049_All41_Original_Loaded_Templates COMMAND test_gk1_scenario_factory ${CMAKE_CURRENT_SOURCE_DIR}/GK1_ALL_41_COUNTRY_ECONOMIC_DATA_v1.07.csv)
add_test(NAME GK1_049_Mongol_Template_CLI COMMAND gk1_scenario_list ${CMAKE_CURRENT_SOURCE_DIR}/GK1_ALL_41_COUNTRY_ECONOMIC_DATA_v1.07.csv 1)
add_test(NAME GK1_049_World_Template_CLI COMMAND gk1_scenario_list ${CMAKE_CURRENT_SOURCE_DIR}/GK1_ALL_41_COUNTRY_ECONOMIC_DATA_v1.07.csv 2)
