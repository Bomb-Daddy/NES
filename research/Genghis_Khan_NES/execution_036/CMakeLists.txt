cmake_minimum_required(VERSION 3.16)
project(gk1_capture_pair C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(gk1_tactical_slice STATIC ../execution_034/gk1_tactical_slice.c)
add_library(gk1_native_ram_view STATIC ../execution_035/gk1_native_ram_view.c)
add_library(gk1_capture_pair STATIC gk1_capture_pair.c)
target_link_libraries(gk1_capture_pair PUBLIC gk1_tactical_slice gk1_native_ram_view)
add_executable(test_gk1_capture_pair test_gk1_capture_pair.c)
target_link_libraries(test_gk1_capture_pair PRIVATE gk1_capture_pair)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  foreach(target gk1_tactical_slice gk1_native_ram_view gk1_capture_pair test_gk1_capture_pair)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
  endforeach()
endif()
enable_testing()
add_test(NAME exec0036_capture_pair COMMAND test_gk1_capture_pair)
