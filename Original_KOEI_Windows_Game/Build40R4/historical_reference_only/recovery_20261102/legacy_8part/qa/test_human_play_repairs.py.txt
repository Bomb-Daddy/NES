def test_d01_map_41_and_farther_click():
    for pid in range(1,42):
        x,y,w,h=SLOTS[str(pid)]['rect_px']
        center=((x+w/2)*960/1448,(y+h/2)*720/1086)
        assert province_at_pixel(*center)==pid
    assert province_at_pixel(970,200) is None
    assert province_at_pixel(-1,10) is None
    # Province 1 is isolated enough to include a click 25px away from marker.
    x,y,w,h=SLOTS['1']['rect_px']
    center=((x+w/2)*960/1448,(y+h/2)*720/1086)
    assert province_at_pixel(center[0]+25,center[1])==1

def test_d07_manual_slot_not_overwritten_by_auto_advance(game):
    old_hash=game.c.fingerprint;old_month=game.c.state['month'];owner=game.player
    game.save()
    manual=game.manual_path();assert manual.exists()
    assert game.path!=manual
    game.month()
    assert game.c.state['month']==old_month+1
    assert game.path.exists() and manual.exists()
    assert game.load() is True
    assert game.c.state['month']==old_month
    assert game.c.fingerprint==old_hash
    assert game.player==owner

def test_d08_event_dialog_blocks_load_without_mutating(game):
    import tkinter as tk
    game.save()
    modal=tk.Toplevel(game.root);game.event_dialog=modal
    before=game.c.fingerprint
    assert game.load() is False
    assert game.c.fingerprint==before
    assert 'Dismiss' in game.msg.get()
    modal.destroy();game.event_dialog=None
    assert game.load() is True

def test_d02_d03_d04_quick_buttons_display_actor_and_preview(game):
    before=game.c.fingerprint
    game.cultivate()
    dialog=game.orders_dialog;game.root.update()
    assert dialog.order.get()=='Cultivate'
    assert not dialog.actor.get()  # Explicit officer selection, no silent default
    dialog.on_nes_button('right')  # D-pad selects first eligible officer
    assert dialog.actor_labels[dialog.actor.get()].startswith(officer_name(game.c.state,dialog.actor.get()))
    assert 'BP ' in dialog.actor_display.get()
    assert game.c.fingerprint==before
    dialog.submit()
    assert dialog.pending is not None
    assert 'Body Points' in dialog.info.get()
    assert game.c.fingerprint==before
    dialog.on_nes_button('b')
    assert game.c.fingerprint==before
    assert dialog.pending is None
    dialog.close()

def test_d15_player_ruler_choice_save_and_replay(game):
    original=game.player
    choices=sorted(game.c.state['factions'],key=lambda x:int(x[1:]))
    assert len(choices)==15
    target=next(x for x in choices if x!=original)
    game.choose_ruler()
    i=choices.index(target)
    game.ruler_listbox.selection_clear(0,'end')
    game.ruler_listbox.selection_set(i)
    game.ruler_confirm()
    assert game.player==target
    # Ruler choice save uses autosave; manual save must also preserve identity.
    game.save()
    game._adopt_loaded(__import__('third_branch.integrated',fromlist=['load_file']).load_file(game.manual_path()))
    assert game.player==target

def test_d06_event_popup_shows_no_code_dict(game):
    event={'kind':'epidemic','province_id':'p12','month':1,
           'result':{'text':'An epidemic has affected local households.',
                     'effects':{'support_ticks':-25000,'unassigned_workers_lost':2}}}
    game._show_committed_events([event])
    game.root.update()
    win=game.event_dialog
    assert win is not None and win.winfo_exists()
    labels=[x.cget('text') for x in win.winfo_children() if x.winfo_class()=='Label' and 'text' in x.keys()]
    shown=' '.join(labels)
    assert 'Province 12' in shown and 'Public support' in shown
    assert "{'" not in shown and 'p12' not in shown and 'support_ticks' not in shown
    game._close_event_dialog()

def test_d03_cancelled_or_repeated_preview_cannot_commit(game):
    game.cultivate()
    dialog=game.orders_dialog
    before=game.c.fingerprint
    dialog.confirm_order()
    assert game.c.fingerprint==before
    dialog.actor.set(dialog.available_officers[0]);dialog.submit()
    assert dialog.pending is not None
    dialog.submit()  # a double click on Preview never commits
    assert game.c.fingerprint==before
    dialog.on_nes_button('b')
    assert dialog.pending is None and game.c.fingerprint==before
    dialog.close()
